cmake_minimum_required(VERSION 3.21)
project(TamberShell VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Quick)
qt_standard_project_setup(REQUIRES 6.5)
qt_add_executable(tamber-shell main.cpp)
qt_add_qml_module(tamber-shell URI TamberShell VERSION 1.0 QML_FILES Main.qml)
target_link_libraries(tamber-shell PRIVATE Qt6::Quick)
install(TARGETS tamber-shell RUNTIME DESTINATION bin)
