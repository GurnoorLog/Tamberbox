cmake_minimum_required(VERSION 3.21)
project(TamberShell VERSION 0.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Quick Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
qt_standard_project_setup(REQUIRES 6.4)
qt_add_executable(tamber-shell main.cpp Launcher.cpp Launcher.h)
qt_add_qml_module(tamber-shell URI TamberShell VERSION 1.0 QML_FILES Main.qml)
qt_add_resources(tamber-shell brand PREFIX "/brand" BASE "../../branding" FILES "../../branding/tamber-llama.svg")
target_link_libraries(tamber-shell PRIVATE Qt6::Quick PkgConfig::GIO)
install(TARGETS tamber-shell RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
 qt_add_executable(launcher-test tests/launcher_test.cpp Launcher.cpp Launcher.h)
 target_include_directories(launcher-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_link_libraries(launcher-test PRIVATE Qt6::Test PkgConfig::GIO)
 add_test(NAME launcher COMMAND launcher-test)
endif()

if(BUILD_TESTING)
 qt_add_executable(launch-probe tests/launch_probe.cpp)
 target_link_libraries(launch-probe PRIVATE Qt6::Quick)
 qt_add_executable(shell-ui-test tests/shell_ui_test.cpp Launcher.cpp Launcher.h)
 target_include_directories(shell-ui-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_link_libraries(shell-ui-test PRIVATE Qt6::Test Qt6::Quick PkgConfig::GIO)
 target_compile_definitions(shell-ui-test PRIVATE TAMBER_QML_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/Main.qml")
 qt_add_resources(shell-ui-test brand PREFIX "/brand" BASE "../../branding" FILES "../../branding/tamber-llama.svg")
 add_test(NAME shell-ui COMMAND shell-ui-test)
endif()

option(TAMBER_WAYLAND_SESSION "Build the Sway/layer-shell desktop" OFF)
if(TAMBER_WAYLAND_SESSION)
 find_package(Qt6 REQUIRED COMPONENTS Network)
 find_package(LayerShellQt REQUIRED)
 qt_add_executable(tamber-desktop desktop/main.cpp desktop/Session.cpp desktop/Session.h Launcher.cpp Launcher.h)
 target_include_directories(tamber-desktop PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_link_libraries(tamber-desktop PRIVATE Qt6::Quick Qt6::Network LayerShellQt::Interface PkgConfig::GIO)
 qt_add_resources(tamber-desktop desktop PREFIX "/desktop" BASE desktop FILES desktop/Desktop.qml)
 qt_add_resources(tamber-desktop artwork PREFIX "/brand" BASE "../../branding" FILES "../../branding/tamber-llama.svg" "../../branding/wallpapers/quiet-ridge.svg")
 install(TARGETS tamber-desktop RUNTIME DESTINATION bin)
endif()
if(TAMBER_WAYLAND_SESSION AND BUILD_TESTING)
 qt_add_executable(session-test tests/session_test.cpp desktop/Session.cpp desktop/Session.h)
 target_include_directories(session-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_link_libraries(session-test PRIVATE Qt6::Test Qt6::Quick LayerShellQt::Interface)
 add_test(NAME session COMMAND session-test)
endif()
